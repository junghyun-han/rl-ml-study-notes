```python
"""3교시 진행 순서(06 문서 0장)에서 쓰는 도구 함수.
환경 확인 → 무작위 기준선 → 정책 평가 → 시드 요약 → 실험 기록
"""
import time
import numpy as np
import gymnasium as gym


# ── W1 환경 확인: 이산/연속, 판 길이, 1,000스텝 걸리는 시간 ──
def inspect_env(env_id):
    env = gym.make(env_id)
    env.reset(seed=0)
    env.action_space.seed(0)
    kind = "이산" if isinstance(env.action_space, gym.spaces.Discrete) else "연속"
    print("관측 공간:", env.observation_space)
    print("행동 공간:", env.action_space, "→", kind)
    print("판 최대 길이:", env.spec.max_episode_steps)
    t0 = time.time()
    for _ in range(1000):
        _, _, term, trunc, _ = env.step(env.action_space.sample())
        if term or trunc:
            env.reset()
    print(f"1,000스텝(무작위 행동) {time.time() - t0:.2f}초")
    return kind


# ── W2 정책 평가: 탐험 없는 정책으로 여러 판, 평균·표준편차·평균 길이 ──
def evaluate(policy_fn, env_id, episodes=10, seed=1000):
    env = gym.make(env_id)
    scores, lengths = [], []
    for i in range(episodes):
        obs, _ = env.reset(seed=seed + i)
        done, total, n = False, 0.0, 0
        while not done:
            obs, r, term, trunc, _ = env.step(policy_fn(obs))
            total += r
            n += 1
            done = term or trunc
        scores.append(total)
        lengths.append(n)
    return float(np.mean(scores)), float(np.std(scores)), float(np.mean(lengths))


# ── W3 무작위 기준선: 학습 전 "바닥 점수" ──
def random_baseline(env_id, episodes=10):
    space = gym.make(env_id).action_space
    space.seed(0)
    return evaluate(lambda obs: space.sample(), env_id, episodes)


# ── W4 시드 여러 개 결과 요약 ──
def summarize(results):
    r = np.asarray(results, dtype=float)
    return f"{r.mean():.1f} ± {r.std():.1f} (시드 {len(r)}개)"


# ── W5 실험 기록: 한 줄씩 표로 남기기 ──
def log_result(table, name, change, score_text, note=""):
    table.append((name, change, score_text, note))
    print(f"| {name} | {change} | {score_text} | {note} |")
```

- **줄별 설명**: 같은 구조인 `05_ppo_discrete.py`가 03 실습노트 7장, 풀이는 08 답안 후보 M1


▸ `mock_ppo.py` 1–97줄
```python
"""모의 실습 M1: 이산 행동 환경에서 PPO로 성능 올리기 (시험장에서 그대로 칠 수 있는 한 파일 풀이)
실행 예: python mock_ppo.py --env CartPole-v1 --steps 60000 --seed 0
"""
import argparse
import numpy as np
import torch
import torch.nn as nn
import gymnasium as gym
from torch.distributions import Categorical


def evaluate(policy_fn, env_id, episodes=10, seed=1000):
    """탐험 없는 정책으로 여러 판 평가 → (평균, 표준편차, 평균 길이). 학습과 다른 시드 사용"""
    env = gym.make(env_id)
    scores, lengths = [], []
    for i in range(episodes):
        obs, _ = env.reset(seed=seed + i)
        done, total, n = False, 0.0, 0
        while not done:
            obs, r, term, trunc, _ = env.step(policy_fn(obs))
            total += r; n += 1
            done = term or trunc
        scores.append(total); lengths.append(n)
    return float(np.mean(scores)), float(np.std(scores)), float(np.mean(lengths))


p = argparse.ArgumentParser()
p.add_argument("--env", default="CartPole-v1")
p.add_argument("--steps", type=int, default=60_000)
p.add_argument("--seed", type=int, default=0)
p.add_argument("--lr", type=float, default=3e-4)
p.add_argument("--ent", type=float, default=0.01)
p.add_argument("--save", default="")          # 판 리턴 기록 파일(곡선 그리기용)
args = p.parse_args()

N_STEPS, EPOCHS, MB, GAMMA, LAM, CLIP, VF, MAX_GRAD = 2048, 10, 64, 0.99, 0.95, 0.2, 0.5, 0.5
torch.manual_seed(args.seed); np.random.seed(args.seed)
env = gym.make(args.env)
n_obs, n_act = env.observation_space.shape[0], env.action_space.n


def layer(i, o, std=np.sqrt(2)):
    l = nn.Linear(i, o); nn.init.orthogonal_(l.weight, std); nn.init.zeros_(l.bias); return l


actor = nn.Sequential(layer(n_obs, 64), nn.Tanh(), layer(64, 64), nn.Tanh(), layer(64, n_act, 0.01))
critic = nn.Sequential(layer(n_obs, 64), nn.Tanh(), layer(64, 64), nn.Tanh(), layer(64, 1, 1.0))
opt = torch.optim.Adam(list(actor.parameters()) + list(critic.parameters()), lr=args.lr, eps=1e-5)

obs, _ = env.reset(seed=args.seed)
ep_ret, returns, step = 0.0, [], 0
for update in range(args.steps // N_STEPS):
    # ① 지금 정책으로 N_STEPS 모으기
    O, A, LP, R, D, V = [], [], [], [], [], []
    for _ in range(N_STEPS):
        o = torch.as_tensor(obs, dtype=torch.float32)
        with torch.no_grad():
            dist = Categorical(logits=actor(o)); a = dist.sample(); v = critic(o).squeeze()
        nxt, r, term, trunc, _ = env.step(a.item()); step += 1; ep_ret += r
        if trunc and not term:                       # 시간 잘림 → 미래 가치를 보상에 더해 부트스트랩
            with torch.no_grad():
                r += GAMMA * critic(torch.as_tensor(nxt, dtype=torch.float32)).item()
        O.append(o); A.append(a); LP.append(dist.log_prob(a)); R.append(r); D.append(float(term or trunc)); V.append(v)
        obs = nxt
        if term or trunc:
            returns.append(ep_ret); ep_ret = 0.0; obs, _ = env.reset()
    O, A, LP, V = torch.stack(O), torch.stack(A), torch.stack(LP), torch.stack(V)
    R, D = torch.tensor(R), torch.tensor(D)
    # ② GAE (뒤에서부터)
    with torch.no_grad():
        last_v = critic(torch.as_tensor(obs, dtype=torch.float32)).squeeze()
    adv, g = torch.zeros(N_STEPS), 0.0
    for t in reversed(range(N_STEPS)):
        nv = last_v if t == N_STEPS - 1 else V[t + 1]
        delta = R[t] + GAMMA * nv * (1 - D[t]) - V[t]
        g = delta + GAMMA * LAM * (1 - D[t]) * g
        adv[t] = g
    ret = adv + V
    # ③ 같은 데이터로 EPOCHS번, 미니배치 업데이트
    for _ in range(EPOCHS):
        for idx in torch.randperm(N_STEPS).split(MB):
            dist = Categorical(logits=actor(O[idx]))
            ratio = (dist.log_prob(A[idx]) - LP[idx]).exp()
            a_n = (adv[idx] - adv[idx].mean()) / (adv[idx].std() + 1e-8)
            pg = -torch.min(ratio * a_n, ratio.clamp(1 - CLIP, 1 + CLIP) * a_n).mean()
            vl = ((critic(O[idx]).squeeze(-1) - ret[idx]) ** 2).mean()
            loss = pg + VF * vl - args.ent * dist.entropy().mean()
            opt.zero_grad(); loss.backward()
            nn.utils.clip_grad_norm_(list(actor.parameters()) + list(critic.parameters()), MAX_GRAD); opt.step()
    if (update + 1) % 5 == 0:
        print(f"step {step:6d} | 최근 20판 학습 리턴 {np.mean(returns[-20:]):6.1f}")

# ④ 탐험 없이 평가 (argmax 행동, 학습과 다른 시드)
mean, std, length = evaluate(lambda o: int(actor(torch.as_tensor(o, dtype=torch.float32)).argmax()), args.env, episodes=10)
print(f"FINAL seed {args.seed}: 평가 10판 {mean:.1f} ± {std:.1f} (평균 길이 {length:.0f})")
if args.save:
    np.savetxt(args.save, np.array(returns))
```

## C2-1. `mock_ppo_plus.py` — PPO 개선판 한 파일 (이산·연속 자동, Gym·MuJoCo)
- **언제**: "PPO로 성능을 높여라"에서 개선을 하나씩 켜 보며 비교할 때. 행동이 이산이면 Categorical, 연속이면 Normal로 **자동**. 옵션: `--norm-obs`(관측 정규화) `--norm-reward`(보상 정규화) `--lr-decay`(학습률 선형 감소) `--target-kl 0.02`(KL 조기 종료) `--save-best`(최고 시점 모델로 평가) `--ent` `--epochs` `--clip` `--lr`. 로그에 **KL·clip 비율·엔트로피·학습률**이 같이 나와 안정성 해석에 바로 쓴다.
- **실행**:
```bash
python mock_ppo_plus.py --env InvertedPendulum-v5 --steps 100000 --seed 0                       # 개선 전
python mock_ppo_plus.py --env InvertedPendulum-v5 --steps 100000 --seed 0 --norm-obs --norm-reward --lr-decay --target-kl 0.02 --save-best   # 개선 후
```
- **실제 결과**: InvertedPendulum-v5, 10만 스텝, 시드 0 — **개선 전**: 학습 리턴 5만 1천 스텝에 1000 도달 → 7만~8만 스텝에 922로 흔들림 → 최종 평가 1000.0 ± 0.0 / **개선 후**(옵션 전부): 4만 1천 스텝에 1000 도달 → 끝까지 1000 유지, KL 0.001~0.003·clip 비율 0.005~0.025로 작아짐 → 최종 평가 1000.0 ± 0.0. 최종 점수는 같고 차이는 **도달 속도와 흔들림**(시드 1개라 일반화하지 않음). 그 밖에 CartPole·Pendulum(옵션 전부)을 1만 스텝 짧게 돌려 동작 확인
- **줄별 설명**: 뼈대는 C2와 같다. 달라진 곳은 ① `make_env`(원래 점수 기록 → 관측 정규화 → 학습 때만 보상 정규화) ② `policy`·`logp_ent`(이산·연속 분기, 연속은 차원 합산) ③ 업데이트 안의 근사 KL·조기 종료·clip 비율 ④ 평가 때 관측 정규화 값 복사·고정. 11 상황별 행동 코드집 2·5장


▸ `mock_ppo_plus.py` 1–151줄
```python
"""모의 실습 M7: PPO 개선판 한 파일 — 이산·연속 행동 자동 구분, Gym·MuJoCo 모두. 개선 옵션을 하나씩 켜며 비교한다.
실행 예: python mock_ppo_plus.py --env CartPole-v1 --steps 60000
        python mock_ppo_plus.py --env InvertedPendulum-v5 --steps 100000 --norm-obs --norm-reward --lr-decay --target-kl 0.02 --save-best
"""
import argparse
import copy
import numpy as np
import torch
import torch.nn as nn
import gymnasium as gym
from torch.distributions import Categorical, Normal

p = argparse.ArgumentParser()
p.add_argument("--env", default="CartPole-v1")
p.add_argument("--steps", type=int, default=100_000)
p.add_argument("--seed", type=int, default=0)
p.add_argument("--lr", type=float, default=3e-4)
p.add_argument("--ent", type=float, default=0.0)          # 엔트로피 계수 (엔트로피가 금방 0이면 올림)
p.add_argument("--epochs", type=int, default=10)          # 같은 데이터로 학습하는 횟수 (KL이 크면 내림)
p.add_argument("--clip", type=float, default=0.2)         # clip ε
p.add_argument("--norm-obs", action="store_true")         # 관측 정규화
p.add_argument("--norm-reward", action="store_true")      # 보상 정규화 (학습용, 평가는 원래 보상)
p.add_argument("--lr-decay", action="store_true")         # 학습률 선형 감소 → 끝에 0
p.add_argument("--target-kl", type=float, default=0.0)    # 0보다 크면 KL 조기 종료
p.add_argument("--save-best", action="store_true")        # 학습 리턴이 가장 좋았던 시점의 모델로 평가
args = p.parse_args()

N_STEPS, MB, GAMMA, LAM, VF, MAX_GRAD = 2048, 64, 0.99, 0.95, 0.5, 0.5
torch.manual_seed(args.seed); np.random.seed(args.seed)


def make_env(train):
    """원래 점수 기록 → (관측 정규화) → (학습 때만 보상 정규화). 관측 정규화 층도 돌려줌"""
    env = gym.wrappers.RecordEpisodeStatistics(gym.make(args.env))
    obs_norm = None
    if args.norm_obs:
        env = obs_norm = gym.wrappers.NormalizeObservation(env)
    if args.norm_reward and train:
        env = gym.wrappers.NormalizeReward(env, gamma=GAMMA)
    return env, obs_norm


env, obs_norm = make_env(train=True)
disc = isinstance(env.action_space, gym.spaces.Discrete)            # 이산이면 Categorical, 연속이면 Normal
n_obs = env.observation_space.shape[0]
n_out = env.action_space.n if disc else env.action_space.shape[0]
if not disc:
    low, high = env.action_space.low, env.action_space.high


def layer(i, o, std=np.sqrt(2)):
    l = nn.Linear(i, o); nn.init.orthogonal_(l.weight, std); nn.init.zeros_(l.bias); return l


actor = nn.Sequential(layer(n_obs, 64), nn.Tanh(), layer(64, 64), nn.Tanh(), layer(64, n_out, 0.01))
critic = nn.Sequential(layer(n_obs, 64), nn.Tanh(), layer(64, 64), nn.Tanh(), layer(64, 1, 1.0))
log_std = nn.Parameter(torch.zeros(n_out))                          # 연속 행동일 때만 사용
params = list(actor.parameters()) + list(critic.parameters()) + ([] if disc else [log_std])
opt = torch.optim.Adam(params, lr=args.lr, eps=1e-5)
n_updates = args.steps // N_STEPS
sched = torch.optim.lr_scheduler.LinearLR(opt, 1.0, 0.0, total_iters=n_updates) if args.lr_decay else None


def policy(o):
    """분포 하나 돌려줌. log_prob·entropy는 연속이면 행동 차원으로 합산"""
    out = actor(o)
    return Categorical(logits=out) if disc else Normal(out, log_std.exp())


def logp_ent(dist, a):
    if disc:
        return dist.log_prob(a), dist.entropy()
    return dist.log_prob(a).sum(-1), dist.entropy().sum(-1)


obs, _ = env.reset(seed=args.seed)
returns, step, best, best_state = [], 0, -np.inf, None
for update in range(n_updates):
    # ① 지금 정책으로 N_STEPS 모으기
    O, A, LP, R, D, V = [], [], [], [], [], []
    for _ in range(N_STEPS):
        o = torch.as_tensor(obs, dtype=torch.float32)
        with torch.no_grad():
            dist = policy(o); a = dist.sample(); lp, _ = logp_ent(dist, a); v = critic(o).squeeze()
        act = a.item() if disc else np.clip(a.numpy(), low, high)      # 환경에 넣을 때만 범위로 자름
        nxt, r, term, trunc, info = env.step(act); step += 1
        if trunc and not term:                                          # 시간 잘림 → 미래 가치로 부트스트랩
            with torch.no_grad():
                r += GAMMA * critic(torch.as_tensor(nxt, dtype=torch.float32)).item()
        O.append(o); A.append(a); LP.append(lp); R.append(float(r)); D.append(float(term or trunc)); V.append(v)
        obs = nxt
        if term or trunc:
            returns.append(float(info["episode"]["r"]))                 # 정규화 전 원래 점수
            obs, _ = env.reset()
    O, A, LP, V = torch.stack(O), torch.stack(A), torch.stack(LP), torch.stack(V)
    R, D = torch.tensor(R), torch.tensor(D)
    # ② GAE (뒤에서부터)
    with torch.no_grad():
        last_v = critic(torch.as_tensor(obs, dtype=torch.float32)).squeeze()
    adv, g = torch.zeros(N_STEPS), 0.0
    for t in reversed(range(N_STEPS)):
        nv = last_v if t == N_STEPS - 1 else V[t + 1]
        delta = R[t] + GAMMA * nv * (1 - D[t]) - V[t]
        g = delta + GAMMA * LAM * (1 - D[t]) * g
        adv[t] = g
    ret = adv + V
    # ③ 미니배치 업데이트 + 진단값(KL·clip 비율·엔트로피) 기록, KL 조기 종료
    kls, clips, ents, stop = [], [], [], False
    for _ in range(args.epochs):
        for idx in torch.randperm(N_STEPS).split(MB):
            new_lp, ent = logp_ent(policy(O[idx]), A[idx])
            log_ratio = new_lp - LP[idx]; ratio = log_ratio.exp()
            kl = ((ratio - 1) - log_ratio).mean().item()               # 근사 KL
            if args.target_kl > 0 and kl > 1.5 * args.target_kl:       # 너무 많이 바뀌면 이번 업데이트 중단
                stop = True; break
            a_n = (adv[idx] - adv[idx].mean()) / (adv[idx].std() + 1e-8)
            pg = -torch.min(ratio * a_n, ratio.clamp(1 - args.clip, 1 + args.clip) * a_n).mean()
            vl = ((critic(O[idx]).squeeze(-1) - ret[idx]) ** 2).mean()
            loss = pg + VF * vl - args.ent * ent.mean()
            opt.zero_grad(); loss.backward(); nn.utils.clip_grad_norm_(params, MAX_GRAD); opt.step()
            kls.append(kl); clips.append(((ratio - 1).abs() > args.clip).float().mean().item()); ents.append(ent.mean().item())
        if stop:
            break
    if sched:
        sched.step()
    recent = np.mean(returns[-20:]) if returns else float("nan")
    if args.save_best and returns and recent > best:                    # 최고 시점 저장 (관측 정규화 값 포함)
        best = recent
        best_state = (copy.deepcopy(actor.state_dict()), log_std.detach().clone(), copy.deepcopy(obs_norm.obs_rms) if obs_norm else None)
    if (update + 1) % 5 == 0:
        print(f"step {step:7d} | 최근 20판 리턴 {recent:8.1f} | KL {np.mean(kls) if kls else 0:.4f} | clip {np.mean(clips) if clips else 0:.3f} | "
              f"엔트로피 {np.mean(ents) if ents else 0:.3f} | lr {opt.param_groups[0]['lr']:.1e}{' | KL 조기 종료' if stop else ''}")

# ④ 탐험 없이 평가 (이산 argmax / 연속 평균 행동), 관측 정규화 값은 복사·고정, 보상은 원래 점수
rms = obs_norm.obs_rms if obs_norm else None
if best_state:
    actor.load_state_dict(best_state[0]); rms = best_state[2]
    print(f"최고 시점 모델로 평가 (학습 리턴 {best:.1f})")
ev, ev_norm = make_env(train=False)
if ev_norm:
    ev_norm.obs_rms = copy.deepcopy(rms); ev_norm.update_running_mean = False
scores, lengths = [], []
for i in range(10):
    o, _ = ev.reset(seed=1000 + i); done, total, n = False, 0.0, 0
    while not done:
        with torch.no_grad():
            out = actor(torch.as_tensor(o, dtype=torch.float32))
        act = int(out.argmax()) if disc else np.clip(out.numpy(), low, high)
        o, r, term, trunc, _ = ev.step(act); total += r; n += 1; done = term or trunc
    scores.append(total); lengths.append(n)
print(f"FINAL seed {args.seed}: 평가 10판 {np.mean(scores):.1f} ± {np.std(scores):.1f} (평균 길이 {np.mean(lengths):.0f})")
```

```python
"""모의 실습 M2: 연속 행동 환경에서 SAC로 성능 올리기 (Pendulum, MuJoCo 등 환경 이름만 바꿔 사용)
실행 예: python mock_sac.py --env Pendulum-v1 --steps 15000 --seed 0
"""
import argparse
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
import gymnasium as gym
from torch.distributions import Normal


def evaluate(policy_fn, env_id, episodes=10, seed=1000):
    """탐험 없는 정책으로 여러 판 평가 → (평균, 표준편차, 평균 길이). 학습과 다른 시드 사용"""
    env = gym.make(env_id)
    scores, lengths = [], []
    for i in range(episodes):
        obs, _ = env.reset(seed=seed + i)
        done, total, n = False, 0.0, 0
        while not done:
            obs, r, term, trunc, _ = env.step(policy_fn(obs))
            total += r; n += 1
            done = term or trunc
        scores.append(total); lengths.append(n)
    return float(np.mean(scores)), float(np.std(scores)), float(np.mean(lengths))


p = argparse.ArgumentParser()
p.add_argument("--env", default="Pendulum-v1")
p.add_argument("--steps", type=int, default=15_000)
p.add_argument("--seed", type=int, default=0)
p.add_argument("--save", default="")          # 판 리턴 기록 파일(곡선 그리기용)
args = p.parse_args()

GAMMA, TAU, LR, BATCH, START, BUF = 0.99, 0.005, 3e-4, 256, 1000, 100_000
torch.manual_seed(args.seed); np.random.seed(args.seed)
env = gym.make(args.env)
n_obs, n_act = env.observation_space.shape[0], env.action_space.shape[0]
low, high = env.action_space.low, env.action_space.high
scale = torch.tensor((high - low) / 2, dtype=torch.float32); bias = torch.tensor((high + low) / 2, dtype=torch.float32)
mlp = lambda i, o: nn.Sequential(nn.Linear(i, 256), nn.ReLU(), nn.Linear(256, 256), nn.ReLU(), nn.Linear(256, o))


class Actor(nn.Module):
    def __init__(self):
        super().__init__(); self.net = mlp(n_obs, 2 * n_act)

    def forward(self, s):
        mu, log_std = self.net(s).chunk(2, -1)
        dist = Normal(mu, log_std.clamp(-20, 2).exp())
        u = dist.rsample(); y = torch.tanh(u)
        logp = dist.log_prob(u).sum(-1) - torch.log(scale * (1 - y.pow(2)) + 1e-6).sum(-1)
        return y * scale + bias, logp, torch.tanh(mu) * scale + bias


actor = Actor()
q1, q2, q1t, q2t = (mlp(n_obs + n_act, 1) for _ in range(4))
q1t.load_state_dict(q1.state_dict()); q2t.load_state_dict(q2.state_dict())
opt_pi = torch.optim.Adam(actor.parameters(), lr=LR)
opt_q = torch.optim.Adam(list(q1.parameters()) + list(q2.parameters()), lr=LR)
log_alpha = torch.zeros(1, requires_grad=True); opt_a = torch.optim.Adam([log_alpha], lr=LR); target_ent = -float(n_act)
S = np.zeros((BUF, n_obs), np.float32); A = np.zeros((BUF, n_act), np.float32)
Rw = np.zeros(BUF, np.float32); S2 = np.zeros((BUF, n_obs), np.float32); T = np.zeros(BUF, np.float32)
ptr = size = 0

obs, _ = env.reset(seed=args.seed); env.action_space.seed(args.seed)
ep_ret, returns = 0.0, []
for step in range(1, args.steps + 1):
    if step < START:
        act = env.action_space.sample()                                   # 처음엔 무작위로 버퍼 채우기
    else:
        with torch.no_grad():
            act = actor(torch.as_tensor(obs, dtype=torch.float32).unsqueeze(0))[0][0].numpy()
    nxt, r, term, trunc, _ = env.step(act)
    S[ptr], A[ptr], Rw[ptr], S2[ptr], T[ptr] = obs, act, r, nxt, float(term)   # terminated만 끝으로
    ptr, size = (ptr + 1) % BUF, min(size + 1, BUF)
    obs, ep_ret = nxt, ep_ret + r
    if term or trunc:
        returns.append(ep_ret); ep_ret = 0.0; obs, _ = env.reset()
    if step >= START:
        i = np.random.randint(0, size, BATCH)
        s, a, rb, s2, d = (torch.as_tensor(x[i]) for x in (S, A, Rw, S2, T))
        alpha = log_alpha.exp().detach()
        with torch.no_grad():                                                # ① Q 목표
            a2, lp2, _ = actor(s2); sa2 = torch.cat([s2, a2], 1)
            y = rb + GAMMA * (1 - d) * (torch.min(q1t(sa2), q2t(sa2)).squeeze(1) - alpha * lp2)
        sa = torch.cat([s, a], 1)
        lq = F.mse_loss(q1(sa).squeeze(1), y) + F.mse_loss(q2(sa).squeeze(1), y)
        opt_q.zero_grad(); lq.backward(); opt_q.step()
        an, lp, _ = actor(s); san = torch.cat([s, an], 1)                     # ② 정책
        lpi = (alpha * lp - torch.min(q1(san), q2(san)).squeeze(1)).mean()
        opt_pi.zero_grad(); lpi.backward(); opt_pi.step()
        la = -(log_alpha.exp() * (lp.detach() + target_ent)).mean()           # ③ 온도 α
        opt_a.zero_grad(); la.backward(); opt_a.step()
        with torch.no_grad():                                                # ④ Polyak
            for net, tgt in ((q1, q1t), (q2, q2t)):
                for p_, pt in zip(net.parameters(), tgt.parameters()):
                    pt.mul_(1 - TAU).add_(TAU * p_)
    if step % 3000 == 0:
        print(f"step {step:6d} | 최근 5판 학습 리턴 {np.mean(returns[-5:]):8.1f} | α {log_alpha.exp().item():.3f}")

# ⑤ 탐험 없이 평가 (평균 행동)
pol = lambda o: actor(torch.as_tensor(o, dtype=torch.float32).unsqueeze(0))[2][0].detach().numpy()
mean, std, length = evaluate(pol, args.env, episodes=10)
print(f"FINAL seed {args.seed}: 평가 10판 {mean:.1f} ± {std:.1f} (평균 길이 {length:.0f})")
if args.save:
    np.savetxt(args.save, np.array(returns))
```

```python
"""06. PPO 연속 행동 버전 — Pendulum-v1
05_ppo_discrete.py와 달라지는 곳만 바꿈: Categorical → Normal(가우시안), log_prob·entropy를 행동 차원으로 합산, 행동 범위 자르기
실행: python 06_ppo_continuous.py
"""
import numpy as np
import torch
import torch.nn as nn
import gymnasium as gym
from torch.distributions import Normal

TOTAL, N_STEPS, EPOCHS, MINIBATCH = 200_000, 2048, 10, 64
GAMMA, LAM, CLIP, LR, ENT_COEF, VF_COEF, MAX_GRAD = 0.99, 0.95, 0.2, 3e-4, 0.0, 0.5, 0.5
SEED = 0
torch.manual_seed(SEED)

env = gym.make("Pendulum-v1")
env = gym.wrappers.NormalizeReward(env, gamma=GAMMA)    # 보상 크기 정규화 (학습 안정)
obs_dim, act_dim = env.observation_space.shape[0], env.action_space.shape[0]
low, high = env.action_space.low, env.action_space.high  # [-2, 2]


class ActorCritic(nn.Module):
    def __init__(self):
        super().__init__()
        self.mu = nn.Sequential(nn.Linear(obs_dim, 64), nn.Tanh(), nn.Linear(64, 64), nn.Tanh(), nn.Linear(64, act_dim))
        self.log_std = nn.Parameter(torch.zeros(act_dim))     # 상태와 무관한 학습 가능한 표준편차
        self.critic = nn.Sequential(nn.Linear(obs_dim, 64), nn.Tanh(), nn.Linear(64, 64), nn.Tanh(), nn.Linear(64, 1))

    def value(self, x):
        return self.critic(x).squeeze(-1)

    def act(self, x, action=None):
        dist = Normal(self.mu(x), self.log_std.exp())
        if action is None:
            action = dist.sample()
        return action, dist.log_prob(action).sum(-1), dist.entropy().sum(-1)   # 차원별 값을 합산


model = ActorCritic()
opt = torch.optim.Adam(model.parameters(), lr=LR, eps=1e-5)
b_obs, b_act = torch.zeros(N_STEPS, obs_dim), torch.zeros(N_STEPS, act_dim)
b_logp, b_rew, b_done, b_val = (torch.zeros(N_STEPS) for _ in range(4))

obs, _ = env.reset(seed=SEED)
log, step = [], 0
for update in range(TOTAL // N_STEPS):
    # ① 롤아웃 모으기
    for t in range(N_STEPS):
        o = torch.as_tensor(obs, dtype=torch.float32)
        with torch.no_grad():
            a, logp, _ = model.act(o)
            v = model.value(o)
        a_env = np.clip(a.numpy(), low, high)              # 환경에는 범위 안으로 자른 값을 보냄
        next_obs, r, terminated, truncated, info = env.step(a_env)
        step += 1
        if truncated and not terminated:
            with torch.no_grad():
                r += GAMMA * model.value(torch.as_tensor(next_obs, dtype=torch.float32)).item()
        b_obs[t], b_act[t], b_logp[t], b_val[t] = o, a, logp, v   # 버퍼에는 자르기 전 a (log_prob 일치)
        b_rew[t], b_done[t] = r, float(terminated or truncated)
        obs = next_obs
        if terminated or truncated:
            obs, _ = env.reset()

    # ② GAE (05와 같음)
    with torch.no_grad():
        last_v = model.value(torch.as_tensor(obs, dtype=torch.float32))
    adv, gae = torch.zeros(N_STEPS), 0.0
    for t in reversed(range(N_STEPS)):
        next_v = last_v if t == N_STEPS - 1 else b_val[t + 1]
        nonterm = 1.0 - b_done[t]
        delta = b_rew[t] + GAMMA * next_v * nonterm - b_val[t]
        gae = delta + GAMMA * LAM * nonterm * gae
        adv[t] = gae
    ret = adv + b_val

    # ③ PPO 업데이트 (05와 같음)
    for epoch in range(EPOCHS):
        perm = torch.randperm(N_STEPS)
        for start in range(0, N_STEPS, MINIBATCH):
            idx = perm[start:start + MINIBATCH]
            _, new_logp, entropy = model.act(b_obs[idx], b_act[idx])
            ratio = (new_logp - b_logp[idx]).exp()
            mb_adv = (adv[idx] - adv[idx].mean()) / (adv[idx].std() + 1e-8)
            policy_loss = -torch.min(ratio * mb_adv, torch.clamp(ratio, 1 - CLIP, 1 + CLIP) * mb_adv).mean()
            value_loss = ((model.value(b_obs[idx]) - ret[idx]) ** 2).mean()
            loss = policy_loss + VF_COEF * value_loss - ENT_COEF * entropy.mean()
            opt.zero_grad(); loss.backward(); nn.utils.clip_grad_norm_(model.parameters(), MAX_GRAD); opt.step()

    # 평가: 정규화 안 된 원래 보상으로, 평균 행동(탐험 없음) 3판
    ev_env, scores = gym.make("Pendulum-v1"), []
    for i in range(3):
        o2, _ = ev_env.reset(seed=100 + i); done, tot = False, 0.0
        while not done:
            with torch.no_grad():
                a_ = model.mu(torch.as_tensor(o2, dtype=torch.float32)).numpy()
            o2, r_, te, tr, _ = ev_env.step(np.clip(a_, low, high)); tot += r_; done = te or tr
        scores.append(tot)
    log.append((step, np.mean(scores)))
    if (update + 1) % 10 == 0:
        print(f"update {update + 1:3d} | step {step:6d} | 평가 평균 {np.mean(scores):8.1f} | std {model.log_std.exp().mean().item():.3f}")

np.savetxt("results/ppo_continuous.txt", np.array(log))
```

- **실제 결과**: 8개 항목 모두 통과
- **줄별 설명**: 11 상황별 행동 코드집


▸ `action_toolbox.py` 1–111줄
```python
"""상황별 행동 코드집에서 쓰는 도구 (모두 실행 확인: python action_toolbox_check.py)
정규화 켜기 · 평가용 환경(정규화 값 복사·고정) · 학습률 바꾸기/줄이기 · PPO 근사 KL · 모델 저장/불러오기 · 시드 여러 개 실행 · 학습 곡선
"""
import copy
import re
import subprocess
import sys
import numpy as np
import torch
import gymnasium as gym


# ── T1 환경 만들기: 관측·보상 정규화, 행동 자르기 켜고 끄기 ──
def make_env(env_id, norm_obs=False, norm_reward=False, clip_action=False, gamma=0.99):
    env = gym.make(env_id)
    if clip_action:
        env = gym.wrappers.ClipAction(env)                      # 연속 행동을 범위 안으로 자름
    if norm_obs:
        env = gym.wrappers.NormalizeObservation(env)            # 관측을 이동 평균·분산으로 정규화
    if norm_reward:
        env = gym.wrappers.NormalizeReward(env, gamma=gamma)    # 보상 크기 맞추기(학습용)
    return env


def find_wrapper(env, cls):
    """감싼 층들을 안쪽으로 따라가며 cls 종류의 Wrapper를 찾음"""
    e = env
    while True:
        if isinstance(e, cls):
            return e
        if not hasattr(e, "env"):
            return None
        e = e.env


# ── T2 평가용 환경: 관측 정규화 값을 복사해 고정, 보상 정규화는 끔(원래 점수로 평가) ──
def make_eval_env(env_id, train_env, clip_action=False):
    src = find_wrapper(train_env, gym.wrappers.NormalizeObservation)
    ev = make_env(env_id, norm_obs=src is not None, clip_action=clip_action)
    if src is not None:
        dst = find_wrapper(ev, gym.wrappers.NormalizeObservation)
        dst.obs_rms = copy.deepcopy(src.obs_rms)                # 학습 때 쌓은 평균·분산 복사
        dst.update_running_mean = False                         # 평가 중에는 갱신하지 않음
    return ev


def evaluate_env(env, policy_fn, episodes=10, seed=1000):
    """이미 만든 환경 객체로 평가 (정규화된 관측을 정책에 넣고, 점수는 원래 보상으로)"""
    scores = []
    for i in range(episodes):
        obs, _ = env.reset(seed=seed + i)
        done, total = False, 0.0
        while not done:
            obs, r, term, trunc, _ = env.step(policy_fn(obs))
            total += r
            done = term or trunc
        scores.append(total)
    return float(np.mean(scores)), float(np.std(scores))


# ── T3 학습률 바꾸기 / 학습하며 선형으로 줄이기 ──
def set_lr(opt, lr):
    for g in opt.param_groups:
        g["lr"] = lr


def linear_decay(opt, total_updates):
    """매 업데이트 끝에 sched.step() 호출 → 학습률이 처음 값에서 0까지 선형으로 감소"""
    return torch.optim.lr_scheduler.LinearLR(opt, start_factor=1.0, end_factor=0.0, total_iters=total_updates)


# ── T4 PPO 근사 KL: 미니배치마다 계산해 목표를 넘으면 그 업데이트의 남은 epoch를 멈춤 ──
def approx_kl(new_logp, old_logp):
    log_ratio = new_logp - old_logp
    return ((log_ratio.exp() - 1) - log_ratio).mean().item()
```

```python
"""02. DQN (+ Double DQN) — CartPole-v1
핵심 3가지: ① Q 신경망  ② 경험 재생 버퍼  ③ 타깃 네트워크
실행: python 02_dqn_cartpole.py            (Double DQN 끄기: --no-double, 타깃망 끄기: --no-target)
"""
import argparse
import random
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
import gymnasium as gym

p = argparse.ArgumentParser()
p.add_argument("--steps", type=int, default=50_000)
p.add_argument("--no-double", action="store_true")
p.add_argument("--no-target", action="store_true")
p.add_argument("--seed", type=int, default=0)
p.add_argument("--lr", type=float, default=1e-3)
p.add_argument("--target-every", type=int, default=500)
args = p.parse_args()

# ── 하이퍼파라미터 ──
GAMMA, LR, BATCH = 0.99, args.lr, 128
BUFFER, LEARN_START, TARGET_EVERY = 50_000, 1_000, args.target_every
EPS_START, EPS_END, EPS_DECAY_STEPS = 1.0, 0.05, 15_000

random.seed(args.seed); np.random.seed(args.seed); torch.manual_seed(args.seed)
env = gym.make("CartPole-v1")
obs_dim, n_act = env.observation_space.shape[0], env.action_space.n


class QNet(nn.Module):
    """상태 → 행동마다 Q값 하나씩 (출력 개수 = 행동 개수)"""
    def __init__(self, obs_dim, n_act, hidden=128):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(obs_dim, hidden), nn.ReLU(),
            nn.Linear(hidden, hidden), nn.ReLU(),
            nn.Linear(hidden, n_act))

    def forward(self, x):
        return self.net(x)


class ReplayBuffer:
    """(s, a, r, s', done)을 원형 배열에 저장하고 무작위로 꺼냄"""
    def __init__(self, cap, obs_dim):
        self.s = np.zeros((cap, obs_dim), np.float32)
        self.a = np.zeros(cap, np.int64)
        self.r = np.zeros(cap, np.float32)
        self.s2 = np.zeros((cap, obs_dim), np.float32)
        self.d = np.zeros(cap, np.float32)
        self.cap, self.i, self.size = cap, 0, 0

    def add(self, s, a, r, s2, d):
        self.s[self.i], self.a[self.i], self.r[self.i], self.s2[self.i], self.d[self.i] = s, a, r, s2, d
        self.i = (self.i + 1) % self.cap               # 다 차면 가장 오래된 것부터 덮어씀
        self.size = min(self.size + 1, self.cap)

    def sample(self, n):
        idx = np.random.randint(0, self.size, n)
        return (torch.as_tensor(self.s[idx]), torch.as_tensor(self.a[idx]), torch.as_tensor(self.r[idx]),
                torch.as_tensor(self.s2[idx]), torch.as_tensor(self.d[idx]))


q = QNet(obs_dim, n_act)
q_target = QNet(obs_dim, n_act)
q_target.load_state_dict(q.state_dict())                # 처음엔 똑같이 복사
opt = torch.optim.Adam(q.parameters(), lr=LR)
buf = ReplayBuffer(BUFFER, obs_dim)

obs, _ = env.reset(seed=args.seed)
ep_ret, ep_returns, log = 0.0, [], []
for step in range(1, args.steps + 1):
    # ① ε-greedy 행동 선택 (ε는 선형으로 줄임)
    eps = max(EPS_END, EPS_START - (EPS_START - EPS_END) * step / EPS_DECAY_STEPS)
    if random.random() < eps:
        action = random.randrange(n_act)            # 시드 고정된 난수로 (재현 가능)
    else:
        with torch.no_grad():
            action = int(q(torch.as_tensor(obs)).argmax())

    # ② 환경 한 스텝 + 버퍼 저장
    next_obs, r, terminated, truncated, _ = env.step(action)
    buf.add(obs, action, r, next_obs, float(terminated))   # 시간 잘림(truncated)은 끝이 아님 → 0으로 저장
    obs, ep_ret = next_obs, ep_ret + r
    if terminated or truncated:
        ep_returns.append(ep_ret)
        log.append((step, ep_ret))
        obs, _ = env.reset()
        ep_ret = 0.0

    # ③ 학습: 버퍼에서 무작위 배치
    if step >= LEARN_START:
        s, a, r_b, s2, d = buf.sample(BATCH)
        q_sa = q(s).gather(1, a.unsqueeze(1)).squeeze(1)          # 내가 한 행동의 Q(s,a)
        with torch.no_grad():
            tnet = q if args.no_target else q_target
            if args.no_double:
                next_q = tnet(s2).max(1).values                      # DQN: 타깃망이 고르고 평가
            else:
                a2 = q(s2).argmax(1, keepdim=True)                   # Double: 온라인망이 고르고
                next_q = tnet(s2).gather(1, a2).squeeze(1)           #         타깃망이 평가
            y = r_b + GAMMA * (1 - d) * next_q                        # 끝난 상태면 미래 가치 0
        loss = F.smooth_l1_loss(q_sa, y)                              # Huber 손실
        opt.zero_grad()
        loss.backward()
        nn.utils.clip_grad_norm_(q.parameters(), 10.0)
        opt.step()

    # ④ 타깃망 주기적 복사 (hard update)
    if not args.no_target and step % TARGET_EVERY == 0:
        q_target.load_state_dict(q.state_dict())

    if step % 5000 == 0:
        recent = np.mean(ep_returns[-20:]) if ep_returns else 0
        print(f"step {step:6d} | 최근 20판 평균 {recent:6.1f} | ε {eps:.2f}")

tag = "dqn" + ("_nodouble" if args.no_double else "") + ("_notarget" if args.no_target else "") + f"_s{args.seed}"
np.savetxt(f"results/{tag}.txt", np.array(log))

# ⑤ 평가: 탐험 없이(greedy) 10판
eval_env = gym.make("CartPole-v1")
scores = []
for i in range(10):
    o, _ = eval_env.reset(seed=1000 + i)
    done, total = False, 0.0
    while not done:
        with torch.no_grad():
            o, r, te, tr, _ = eval_env.step(int(q(torch.as_tensor(o)).argmax()))
        total += r
        done = te or tr
    scores.append(total)
print(f"평가(탐험 없음) 10판 평균: {np.mean(scores):.1f} (최대 500)")
```

```python
"""05. PPO (clip) + GAE — CartPole-v1 (이산 행동)
순서: ① 정책으로 N스텝 모으기 → ② GAE로 이점 계산 → ③ 같은 데이터로 K epoch 미니배치 업데이트(clip)
실행: python 05_ppo_discrete.py            (학습률 바꾸기: --lr 3e-3)
"""
import argparse
import numpy as np
import torch
import torch.nn as nn
import gymnasium as gym
from torch.distributions import Categorical

p = argparse.ArgumentParser()
p.add_argument("--steps", type=int, default=100_000)
p.add_argument("--lr", type=float, default=3e-4)
p.add_argument("--clip", type=float, default=0.2)
p.add_argument("--seed", type=int, default=0)
args = p.parse_args()

N_STEPS, EPOCHS, MINIBATCH = 2048, 10, 64         # PPO 논문 MuJoCo 표와 같은 값
GAMMA, LAM, CLIP = 0.99, 0.95, args.clip
ENT_COEF, VF_COEF, MAX_GRAD = 0.01, 0.5, 0.5

torch.manual_seed(args.seed); np.random.seed(args.seed)
env = gym.make("CartPole-v1")
obs_dim, n_act = env.observation_space.shape[0], env.action_space.n


def layer(i, o, std=np.sqrt(2)):
    """직교 초기화 (PPO 구현에서 흔히 쓰는 방식). 정책 마지막 층은 0.01로 작게 → 처음엔 거의 균등 확률"""
    l = nn.Linear(i, o)
    nn.init.orthogonal_(l.weight, std)
    nn.init.zeros_(l.bias)
    return l


class ActorCritic(nn.Module):
    def __init__(self):
        super().__init__()
        self.actor = nn.Sequential(layer(obs_dim, 64), nn.Tanh(), layer(64, 64), nn.Tanh(), layer(64, n_act, 0.01))
        self.critic = nn.Sequential(layer(obs_dim, 64), nn.Tanh(), layer(64, 64), nn.Tanh(), layer(64, 1, 1.0))

    def value(self, x):
        return self.critic(x).squeeze(-1)

    def act(self, x, action=None):
        dist = Categorical(logits=self.actor(x))
        if action is None:
            action = dist.sample()
        return action, dist.log_prob(action), dist.entropy()


model = ActorCritic()
opt = torch.optim.Adam(model.parameters(), lr=args.lr, eps=1e-5)

# 롤아웃 저장 공간
b_obs = torch.zeros(N_STEPS, obs_dim)
b_act = torch.zeros(N_STEPS, dtype=torch.long)
b_logp, b_rew, b_done, b_val = (torch.zeros(N_STEPS) for _ in range(4))

obs, _ = env.reset(seed=args.seed)
ep_ret, log, step = 0.0, [], 0
n_updates = args.steps // N_STEPS
for update in range(n_updates):
    # ① 현재 정책(π_old)으로 N_STEPS 모으기
    for t in range(N_STEPS):
        o = torch.as_tensor(obs, dtype=torch.float32)
        with torch.no_grad():
            a, logp, _ = model.act(o)
            v = model.value(o)
        next_obs, r, terminated, truncated, _ = env.step(a.item())
        step += 1
        ep_ret += r
        if truncated and not terminated:
            # 시간 제한으로 잘린 것은 "끝"이 아님 → 다음 상태 가치를 보상에 더해 부트스트랩
            with torch.no_grad():
                r += GAMMA * model.value(torch.as_tensor(next_obs, dtype=torch.float32)).item()
        b_obs[t], b_act[t], b_logp[t], b_val[t] = o, a, logp, v
        b_rew[t], b_done[t] = r, float(terminated or truncated)
        obs = next_obs
        if terminated or truncated:
            log.append((step, ep_ret))
            ep_ret = 0.0
            obs, _ = env.reset()

    # ② GAE(λ): δ_t = r_t + γV(s_(t+1)) − V(s_t),  Â_t = δ_t + γλ Â_(t+1)
    with torch.no_grad():
        last_v = model.value(torch.as_tensor(obs, dtype=torch.float32))
    adv = torch.zeros(N_STEPS)
    gae = 0.0
    for t in reversed(range(N_STEPS)):
        next_v = last_v if t == N_STEPS - 1 else b_val[t + 1]
        nonterm = 1.0 - b_done[t]                      # 판이 끝났으면 다음 판 값을 끌어오지 않음
        delta = b_rew[t] + GAMMA * next_v * nonterm - b_val[t]
        gae = delta + GAMMA * LAM * nonterm * gae
        adv[t] = gae
    ret = adv + b_val                                  # Critic 학습 목표 (= λ-return)

    # ③ 같은 데이터로 EPOCHS번, 미니배치로 나눠 업데이트
    kls, clipfracs = [], []
    for epoch in range(EPOCHS):
        perm = torch.randperm(N_STEPS)
        for start in range(0, N_STEPS, MINIBATCH):
            idx = perm[start:start + MINIBATCH]
            _, new_logp, entropy = model.act(b_obs[idx], b_act[idx])
            log_ratio = new_logp - b_logp[idx]
            ratio = log_ratio.exp()                                    # r_t(θ) = π_θ / π_old
            mb_adv = (adv[idx] - adv[idx].mean()) / (adv[idx].std() + 1e-8)   # 이점 정규화

            surr1 = ratio * mb_adv
            surr2 = torch.clamp(ratio, 1 - CLIP, 1 + CLIP) * mb_adv
            policy_loss = -torch.min(surr1, surr2).mean()               # L^CLIP (최대화 → 음수)
            value_loss = ((model.value(b_obs[idx]) - ret[idx]) ** 2).mean()
            loss = policy_loss + VF_COEF * value_loss - ENT_COEF * entropy.mean()

            opt.zero_grad()
            loss.backward()
            nn.utils.clip_grad_norm_(model.parameters(), MAX_GRAD)
            opt.step()

            with torch.no_grad():                                       # 진단용 지표
                kls.append(((ratio - 1) - log_ratio).mean().item())     # 근사 KL
                clipfracs.append(((ratio - 1).abs() > CLIP).float().mean().item())

    ev = 1 - torch.var(ret - b_val) / (torch.var(ret) + 1e-8)          # 설명 분산 (1에 가까울수록 Critic 좋음)
    recent = np.mean([r for _, r in log[-20:]]) if log else 0
    print(f"update {update + 1:3d} | step {step:6d} | 최근 20판 {recent:6.1f} | "
          f"KL {np.mean(kls):.4f} | clipfrac {np.mean(clipfracs):.3f} | entropy {entropy.mean().item():.3f} | EV {ev:.2f}")

np.savetxt(f"results/ppo_lr{args.lr:g}_clip{args.clip:g}_s{args.seed}.txt", np.array(log))
```

## C8. `07_sac_pendulum.py` — SAC, Pendulum (노트 버전)
- **언제**: C3과 같은 SAC를 줄별 설명과 함께 볼 때.

```bash
python 07_sac_pendulum.py
```

- **줄별 설명**: 03 실습노트 9장


▸ `07_sac_pendulum.py` 1–125줄
```python
"""07. SAC (Soft Actor-Critic) — Pendulum-v1 (연속 행동)
요즘 표준 구현(후속 논문 1812.05905 방식): Q 두 개 + 타깃 Q(Polyak) + 온도 α 자동 조절.
원 논문(1801.01290)과의 차이: 원 논문은 V 네트워크와 타깃 V를 따로 두고, α 대신 보상 크기(reward scale)를 조절.
실행: python 07_sac_pendulum.py
"""
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
import gymnasium as gym
from torch.distributions import Normal

TOTAL, LEARN_START, BATCH, BUFFER = 15_000, 1_000, 256, 100_000
GAMMA, TAU, LR = 0.99, 0.005, 3e-4                  # 원 논문 표 1과 같은 값
SEED = 0
torch.manual_seed(SEED); np.random.seed(SEED)

env = gym.make("Pendulum-v1")
obs_dim, act_dim = env.observation_space.shape[0], env.action_space.shape[0]
act_scale = torch.tensor((env.action_space.high - env.action_space.low) / 2, dtype=torch.float32)  # 2
act_bias = torch.tensor((env.action_space.high + env.action_space.low) / 2, dtype=torch.float32)   # 0


def mlp(i, o, h=256):
    return nn.Sequential(nn.Linear(i, h), nn.ReLU(), nn.Linear(h, h), nn.ReLU(), nn.Linear(h, o))


class Actor(nn.Module):
    """가우시안 정책 + tanh로 범위 제한"""
    def __init__(self):
        super().__init__()
        self.body = nn.Sequential(nn.Linear(obs_dim, 256), nn.ReLU(), nn.Linear(256, 256), nn.ReLU())
        self.mu = nn.Linear(256, act_dim)
        self.log_std = nn.Linear(256, act_dim)

    def forward(self, s):
        h = self.body(s)
        mu, log_std = self.mu(h), self.log_std(h).clamp(-20, 2)   # 표준편차가 너무 작거나 크지 않게
        dist = Normal(mu, log_std.exp())
        u = dist.rsample()                         # 재매개변수화: u = μ + σ·ε  (기울기가 μ, σ로 흐름)
        y = torch.tanh(u)
        action = y * act_scale + act_bias
        # tanh 변환에 따른 확률 보정: log π(a|s) = log N(u) − Σ log(scale·(1 − tanh²u))
        log_prob = dist.log_prob(u).sum(-1) - torch.log(act_scale * (1 - y.pow(2)) + 1e-6).sum(-1)
        mean_action = torch.tanh(mu) * act_scale + act_bias          # 평가용(탐험 없음)
        return action, log_prob, mean_action


actor = Actor()
q1, q2 = mlp(obs_dim + act_dim, 1), mlp(obs_dim + act_dim, 1)
q1_t, q2_t = mlp(obs_dim + act_dim, 1), mlp(obs_dim + act_dim, 1)
q1_t.load_state_dict(q1.state_dict()); q2_t.load_state_dict(q2.state_dict())
opt_actor = torch.optim.Adam(actor.parameters(), lr=LR)
opt_q = torch.optim.Adam(list(q1.parameters()) + list(q2.parameters()), lr=LR)
log_alpha = torch.zeros(1, requires_grad=True)       # α = exp(log_alpha) > 0 이 되도록
opt_alpha = torch.optim.Adam([log_alpha], lr=LR)
target_entropy = -float(act_dim)                     # 목표 엔트로피 = −(행동 차원 수)

# 리플레이 버퍼 (numpy 배열)
S = np.zeros((BUFFER, obs_dim), np.float32); A = np.zeros((BUFFER, act_dim), np.float32)
R = np.zeros(BUFFER, np.float32); S2 = np.zeros((BUFFER, obs_dim), np.float32); T = np.zeros(BUFFER, np.float32)
ptr, size = 0, 0

obs, _ = env.reset(seed=SEED)
ep_ret, log = 0.0, []
for step in range(1, TOTAL + 1):
    if step < LEARN_START:
        action = np.random.uniform(env.action_space.low, env.action_space.high)   # 처음엔 무작위로 버퍼 채우기
    else:
        with torch.no_grad():
            action = actor(torch.as_tensor(obs).unsqueeze(0))[0][0].numpy()
    next_obs, r, terminated, truncated, _ = env.step(action)
    S[ptr], A[ptr], R[ptr], S2[ptr], T[ptr] = obs, action, r, next_obs, float(terminated)
    ptr, size = (ptr + 1) % BUFFER, min(size + 1, BUFFER)
    obs, ep_ret = next_obs, ep_ret + r
    if terminated or truncated:
        log.append((step, ep_ret))
        if len(log) % 10 == 0:
            print(f"step {step:6d} | 최근 10판 평균 {np.mean([x for _, x in log[-10:]]):8.1f} | α {log_alpha.exp().item():.3f}")
        obs, _ = env.reset()
        ep_ret = 0.0

    if step >= LEARN_START:
        idx = np.random.randint(0, size, BATCH)
        s, a, r_b, s2, d = (torch.as_tensor(x[idx]) for x in (S, A, R, S2, T))
        alpha = log_alpha.exp().detach()

        # ① Q 학습: y = r + γ(1−d)[ min(Q1', Q2')(s', a') − α log π(a'|s') ],  a' ~ π(·|s')
        with torch.no_grad():
            a2, logp2, _ = actor(s2)
            sa2 = torch.cat([s2, a2], 1)
            q_next = torch.min(q1_t(sa2), q2_t(sa2)).squeeze(1) - alpha * logp2
            y = r_b + GAMMA * (1 - d) * q_next
        sa = torch.cat([s, a], 1)
        q_loss = F.mse_loss(q1(sa).squeeze(1), y) + F.mse_loss(q2(sa).squeeze(1), y)
        opt_q.zero_grad(); q_loss.backward(); opt_q.step()

        # ② 정책 학습: 최소화  E[ α log π(a|s) − min(Q1, Q2)(s, a) ],  a는 rsample (재매개변수화)
        a_pi, logp, _ = actor(s)
        sa_pi = torch.cat([s, a_pi], 1)
        actor_loss = (alpha * logp - torch.min(q1(sa_pi), q2(sa_pi)).squeeze(1)).mean()
        opt_actor.zero_grad(); actor_loss.backward(); opt_actor.step()

        # ③ 온도 α 자동 조절: 엔트로피가 목표보다 낮으면 α↑ (더 탐험), 높으면 α↓
        alpha_loss = -(log_alpha.exp() * (logp.detach() + target_entropy)).mean()
        opt_alpha.zero_grad(); alpha_loss.backward(); opt_alpha.step()

        # ④ 타깃 Q를 천천히 따라가게 (Polyak 평균): θ' ← τθ + (1−τ)θ'
        with torch.no_grad():
            for net, tgt in ((q1, q1_t), (q2, q2_t)):
                for p_, pt in zip(net.parameters(), tgt.parameters()):
                    pt.mul_(1 - TAU).add_(TAU * p_)

np.savetxt("results/sac.txt", np.array(log))

scores = []
```

```python
"""08. 보상 설계 비교 — 같은 격자, 같은 Q-learning, 보상만 4가지
  sparse    : 도착 +1, 나머지 0                      → 신호가 드물어 느림
  step      : 매 스텝 −0.01, 도착 +1                 → 빨리 도착하도록 재촉
  potential : sparse + γΦ(s') − Φ(s), Φ = −거리      → 빨라지고 최적 정책은 그대로 (Ng 외 1999)
  near_bonus: sparse + 목표 1칸 이내면 매 스텝 +0.1   → 보상 해킹: 도착 안 하고 근처에서 맴돎
실행: python 08_reward_shaping_grid.py
"""
import numpy as np

N, MAX_STEPS, GAMMA = 8, 100, 0.99
GOAL = (N - 1, N - 1)
MOVES = [(-1, 0), (1, 0), (0, -1), (0, 1)]


def dist(s):
    return abs(s[0] - GOAL[0]) + abs(s[1] - GOAL[1])


def step_env(s, a):
    r_, c_ = s[0] + MOVES[a][0], s[1] + MOVES[a][1]
    s2 = (min(max(r_, 0), N - 1), min(max(c_, 0), N - 1))
    return s2, s2 == GOAL                                  # (다음 상태, 도착=종료)


def reward(mode, s, s2, done):
    base = 1.0 if done else 0.0
    if mode == "sparse":
        return base
    if mode == "step":
        return base - 0.01
    if mode == "potential":
        phi = lambda x: -dist(x) / (2 * (N - 1))          # 거리를 0~−1로 맞춘 잠재함수
        phi_next = 0.0 if done else phi(s2)                # 끝난 상태의 Φ = 0
        return base + GAMMA * phi_next - phi(s)
    if mode == "near_bonus":
        return base + (0.1 if (not done and dist(s2) <= 1) else 0.0)


def run(mode, episodes=300, alpha=0.5, eps=0.1, seed=0):
    rng = np.random.default_rng(seed)
    Q = np.zeros((N, N, 4))
    first_success = None
    for ep in range(episodes):
        s = (0, 0)
        for t in range(MAX_STEPS):
            a = int(rng.integers(4)) if rng.random() < eps else int(rng.choice(np.flatnonzero(Q[s] == Q[s].max())))
            s2, done = step_env(s, a)
            r = reward(mode, s, s2, done)
            target = r + GAMMA * (0.0 if done else Q[s2].max())   # 시간 잘림은 끝이 아니므로 부트스트랩
            Q[s][a] += alpha * (target - Q[s][a])
            s = s2
            if done:
                if first_success is None:
                    first_success = ep
                break
    # 학습 끝난 뒤 탐험 없이 한 판: 도착했나, 몇 스텝?
    s, steps, reached = (0, 0), 0, False
    for t in range(MAX_STEPS):
        s, done = step_env(s, int(np.argmax(Q[s])))
        steps += 1
        if done:
            reached = True
            break
    return first_success, reached, steps


if __name__ == "__main__":
    print(f"{N}×{N} 격자, 최단 거리 {2 * (N - 1)}스텝, 시드 5개")
    for mode in ["sparse", "step", "potential", "near_bonus"]:
        res = [run(mode, seed=s) for s in range(5)]
        fs = [r[0] for r in res]
        print(f"{mode:10s} | 처음 도착한 판: {fs} | 학습 후 도착: {[r[1] for r in res]} | 학습 후 스텝: {[r[2] for r in res]}")
```

```python
"""09. RRT와 RRT* — 2D 평면 경로 탐색 (numpy만 사용)
RRT : 무작위 점을 찍고 → 가장 가까운 노드에서 → 그쪽으로 한 걸음(step) → 충돌 없으면 트리에 추가
RRT*: + 주변 반경 안에서 비용이 가장 싼 부모 고르기 + 주변 노드 다시 잇기(rewire) → 경로가 점점 짧아짐
실행: python 09_rrt.py        (그림 저장: results/rrt.png)
"""
import numpy as np

W, H = 10.0, 10.0
OBSTACLES = [(2, 0, 1, 6), (5, 4, 1, 6), (7.5, 0, 1, 6)]   # 사각형 (x, y, 너비, 높이)
START, GOAL = np.array([0.5, 0.5]), np.array([9.5, 9.5])


def point_free(p, margin=0.1):
    if not (0 <= p[0] <= W and 0 <= p[1] <= H):
        return False
    for (x, y, w, h) in OBSTACLES:
        if x - margin <= p[0] <= x + w + margin and y - margin <= p[1] <= y + h + margin:
            return False
    return True


def edge_free(p, q, res=0.05):
    """두 점 사이 선분을 잘게 나눠 모든 점이 장애물 밖인지 확인"""
    n = max(2, int(np.ceil(np.linalg.norm(q - p) / res)))
    return all(point_free(p + (q - p) * t) for t in np.linspace(0, 1, n))


def path_from(nodes, parent, i):
    path = []
    while i != -1:
        path.append(nodes[i])
        i = parent[i]
    return np.array(path[::-1])


def path_length(path):
    return float(np.sum(np.linalg.norm(np.diff(path, axis=0), axis=1)))


def rrt(step=0.5, goal_bias=0.1, goal_tol=0.5, max_iter=5000, seed=0):
    rng = np.random.default_rng(seed)
    nodes, parent = [START.copy()], [-1]
    for it in range(max_iter):
        # ① 샘플: 가끔(goal_bias)은 목표점 자체를 찍어서 목표 쪽으로 끌어당김
        x_rand = GOAL.copy() if rng.random() < goal_bias else rng.uniform([0, 0], [W, H])
        # ② 가장 가까운 노드
        arr = np.array(nodes)
        i_near = int(np.argmin(np.linalg.norm(arr - x_rand, axis=1)))
        x_near = arr[i_near]
        # ③ 그쪽으로 최대 step만큼 이동 (steer)
        d = x_rand - x_near
        dlen = np.linalg.norm(d)
        if dlen < 1e-9:
            continue
        x_new = x_near + d / dlen * min(step, dlen)
        # ④ 충돌 검사 후 추가
        if not edge_free(x_near, x_new):
            continue
        nodes.append(x_new)
        parent.append(i_near)
        # ⑤ 목표에 충분히 가까우면 연결하고 끝
        if np.linalg.norm(x_new - GOAL) <= goal_tol and edge_free(x_new, GOAL):
            nodes.append(GOAL.copy())
            parent.append(len(nodes) - 2)
            return path_from(nodes, parent, len(nodes) - 1), nodes, parent, it + 1
    return None, nodes, parent, max_iter


def rrt_star(step=0.5, goal_bias=0.1, goal_tol=0.5, radius=1.2, max_iter=3000, seed=0):
    rng = np.random.default_rng(seed)
    nodes, parent, cost = [START.copy()], [-1], [0.0]
    goal_idx = None
    for it in range(max_iter):
        x_rand = GOAL.copy() if rng.random() < goal_bias else rng.uniform([0, 0], [W, H])
        arr = np.array(nodes)
        dists = np.linalg.norm(arr - x_rand, axis=1)
        i_near = int(np.argmin(dists))
        d = x_rand - arr[i_near]
        dlen = np.linalg.norm(d)
        if dlen < 1e-9:
            continue
        x_new = arr[i_near] + d / dlen * min(step, dlen)
        if not edge_free(arr[i_near], x_new):
            continue
        # ① 부모 고르기: 반경 안 이웃 중 (그 노드까지 비용 + 거리)가 가장 작은 것
        near = np.flatnonzero(np.linalg.norm(arr - x_new, axis=1) <= radius)
        best_p, best_c = i_near, cost[i_near] + np.linalg.norm(x_new - arr[i_near])
        for j in near:
            c = cost[j] + np.linalg.norm(x_new - arr[j])
            if c < best_c and edge_free(arr[j], x_new):
                best_p, best_c = int(j), c
        nodes.append(x_new); parent.append(best_p); cost.append(best_c)
        new_i = len(nodes) - 1
        # ② 다시 잇기(rewire): 이웃이 x_new를 거쳐 가는 게 더 싸면 부모를 x_new로 바꿈
        for j in near:
            c = best_c + np.linalg.norm(arr[j] - x_new)
            if c < cost[j] and edge_free(x_new, arr[j]):
                parent[j] = new_i
                cost[j] = c            # (간단 버전: 자손 비용은 즉시 갱신하지 않음)
        # ③ 목표 연결: 찾은 뒤에도 멈추지 않고 계속 → 더 짧은 길로 개선
        if np.linalg.norm(x_new - GOAL) <= goal_tol and edge_free(x_new, GOAL):
            c_goal = best_c + np.linalg.norm(GOAL - x_new)
            if goal_idx is None or c_goal < cost[goal_idx]:
                if goal_idx is None:
                    nodes.append(GOAL.copy()); parent.append(new_i); cost.append(c_goal)
                    goal_idx = len(nodes) - 1
                else:
                    parent[goal_idx], cost[goal_idx] = new_i, c_goal
    path = path_from(nodes, parent, goal_idx) if goal_idx is not None else None
    return path, nodes, parent


if __name__ == "__main__":
    p1, n1, par1, iters = rrt()
    p2, n2, par2 = rrt_star()
    print(f"RRT : {iters}번 반복, 노드 {len(n1)}개, 경로 길이 {path_length(p1):.2f}")
    print(f"RRT*: 3000번 반복, 노드 {len(n2)}개, 경로 길이 {path_length(p2):.2f}")
    try:
        import matplotlib
        matplotlib.use("Agg")
        import matplotlib.pyplot as plt
        from matplotlib.patches import Rectangle
        plt.rcParams["font.family"] = "AppleGothic"
        fig, axes = plt.subplots(1, 2, figsize=(10, 5))
        for ax, (path, nodes, parent, title) in zip(axes, [(p1, n1, par1, "RRT"), (p2, n2, par2, "RRT*")]):
            for (x, y, w, h) in OBSTACLES:
                ax.add_patch(Rectangle((x, y), w, h, color="#888"))
            for i, pi in enumerate(parent):
                if pi >= 0:
                    ax.plot([nodes[i][0], nodes[pi][0]], [nodes[i][1], nodes[pi][1]], color="#9cc3e6", lw=0.6)
            ax.plot(path[:, 0], path[:, 1], color="#d62728", lw=2.2)
            ax.plot(*START, "go", ms=8); ax.plot(*GOAL, "r*", ms=12)
            ax.set_xlim(0, W); ax.set_ylim(0, H); ax.set_aspect("equal")
            ax.set_title(f"{title}: 노드 {len(nodes)}개, 경로 {path_length(path):.2f}")
        plt.tight_layout(); plt.savefig("results/rrt.png", dpi=130)
        print("그림 저장: results/rrt.png")
    except ImportError:
        pass
```

```python
"""01. 표 형태 Q-learning vs SARSA (CliffWalking, Sutton & Barto 예제 6.6)
실행: python 01_q_learning_cliff.py
"""
import numpy as np
import gymnasium as gym


def eps_greedy(Q, s, eps, rng):
    """확률 eps로 무작위 행동(탐험), 나머지는 Q가 가장 큰 행동(활용)."""
    if rng.random() < eps:
        return int(rng.integers(Q.shape[1]))
    return int(np.argmax(Q[s]))


def train(method, episodes=1000, alpha=0.1, gamma=1.0, eps=0.1, seed=0):
    env = gym.make("CliffWalking-v1")            # 4×12 격자, 매 스텝 −1, 절벽 −100(출발점으로)
    rng = np.random.default_rng(seed)
    Q = np.zeros((env.observation_space.n, env.action_space.n))   # 상태 48 × 행동 4
    returns = []
    for ep in range(episodes):
        s, _ = env.reset(seed=seed + ep)
        a = eps_greedy(Q, s, eps, rng)
        G, done = 0.0, False
        while not done:
            s2, r, terminated, truncated, _ = env.step(a)
            G += r
            done = terminated or truncated
            a2 = eps_greedy(Q, s2, eps, rng)       # 다음에 실제로 할 행동
            if method == "sarsa":                  # on-policy: 실제로 할 행동 a2의 Q
                target = r + gamma * Q[s2, a2] * (not terminated)
            else:                                  # Q-learning, off-policy: 가장 좋은 행동의 Q
                target = r + gamma * np.max(Q[s2]) * (not terminated)
            Q[s, a] += alpha * (target - Q[s, a])  # TD 오차만큼 조금 이동
            s, a = s2, a2
        returns.append(G)
    return Q, np.array(returns)


def greedy_path(Q):
    """학습된 Q로 탐험 없이 갔을 때의 길을 격자에 그림."""
    env = gym.make("CliffWalking-v1")
    s, _ = env.reset(seed=0)
    grid = [["." for _ in range(12)] for _ in range(4)]
    for c in range(1, 11):
        grid[3][c] = "C"                           # 절벽
    for _ in range(100):
        r_, c_ = divmod(s, 12)
        grid[r_][c_] = "*"
        s, _, term, trunc, _ = env.step(int(np.argmax(Q[s])))
        if term or trunc:
            break
    grid[3][11] = "G"
    return "\n".join("".join(row) for row in grid)


if __name__ == "__main__":
    for m in ["sarsa", "qlearning"]:
        Q, R = train(m)
        np.savetxt(f"results/cliff_{m}.txt", R)
        print(f"[{m}] 마지막 100판 평균 보상(탐험 포함): {R[-100:].mean():.1f}")
        print(greedy_path(Q), "\n")
```

```python
"""03. REINFORCE (몬테카를로 정책 경사) — CartPole-v1
한 판을 끝까지 해 본 뒤, 잘 된 행동의 확률은 올리고 못 된 행동의 확률은 내린다.
실행: python 03_reinforce_cartpole.py
"""
import numpy as np
import torch
import torch.nn as nn
import gymnasium as gym
from torch.distributions import Categorical

GAMMA, LR, EPISODES, SEED = 0.99, 1e-3, 800, 0
torch.manual_seed(SEED)
env = gym.make("CartPole-v1")
policy = nn.Sequential(nn.Linear(4, 128), nn.ReLU(), nn.Linear(128, 2))   # 출력 = 행동별 logit
opt = torch.optim.Adam(policy.parameters(), lr=LR)

returns_log = []
for ep in range(EPISODES):
    obs, _ = env.reset(seed=SEED + ep)
    log_probs, rewards = [], []
    while True:
        dist = Categorical(logits=policy(torch.as_tensor(obs)))   # logit → 확률 분포
        a = dist.sample()                                         # 확률대로 행동 뽑기 (탐험이 저절로 됨)
        log_probs.append(dist.log_prob(a))                        # log π(a|s) 저장 (기울기 연결됨)
        obs, r, terminated, truncated, _ = env.step(a.item())
        rewards.append(r)
        if terminated or truncated:
            break

    # 뒤에서부터 할인 누적 보상 G_t = r_t + γ G_(t+1)
    G, Gs = 0.0, []
    for r in reversed(rewards):
        G = r + GAMMA * G
        Gs.insert(0, G)
    Gs = torch.tensor(Gs)
    Gs = (Gs - Gs.mean()) / (Gs.std() + 1e-8)      # 정규화 = 기준선(baseline) 역할 → 분산 감소

    loss = -(torch.stack(log_probs) * Gs).sum()     # 경사 "상승"을 위해 음수를 붙여 최소화
    opt.zero_grad()
    loss.backward()
    opt.step()

    returns_log.append(sum(rewards))
    if (ep + 1) % 100 == 0:
        print(f"episode {ep + 1:4d} | 최근 50판 평균 {np.mean(returns_log[-50:]):6.1f}")

np.savetxt("results/reinforce.txt", np.array(returns_log))
```

- **언제**: Actor-Critic을 구현하라고 할 때.

▸ `04_actor_critic_cartpole.py` 1–51줄
```python
"""04. Actor-Critic (1-step TD 이점) — CartPole-v1
Actor = 정책 π(a|s), Critic = 상태 가치 V(s).  이점 A ≈ δ = r + γV(s') − V(s)
실행: python 04_actor_critic_cartpole.py
"""
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
import gymnasium as gym
from torch.distributions import Categorical

GAMMA, LR_A, LR_C, EPISODES, SEED = 0.99, 1e-3, 5e-3, 800, 0
torch.manual_seed(SEED)
env = gym.make("CartPole-v1")
actor = nn.Sequential(nn.Linear(4, 128), nn.ReLU(), nn.Linear(128, 2))
critic = nn.Sequential(nn.Linear(4, 128), nn.ReLU(), nn.Linear(128, 1))
opt_a = torch.optim.Adam(actor.parameters(), lr=LR_A)
opt_c = torch.optim.Adam(critic.parameters(), lr=LR_C)

returns_log = []
for ep in range(EPISODES):
    obs, _ = env.reset(seed=SEED + ep)
    S, A, R, S2, T = [], [], [], [], []
    while True:
        with torch.no_grad():
            a = Categorical(logits=actor(torch.as_tensor(obs))).sample().item()
        obs2, r, terminated, truncated, _ = env.step(a)
        S.append(obs); A.append(a); R.append(r); S2.append(obs2); T.append(float(terminated))
        obs = obs2
        if terminated or truncated:
            break

    S, S2 = torch.as_tensor(np.array(S)), torch.as_tensor(np.array(S2))
    A, R, T = torch.as_tensor(A), torch.as_tensor(R, dtype=torch.float32), torch.as_tensor(T)

    v = critic(S).squeeze(1)                                   # V(s)
    with torch.no_grad():
        target = R + GAMMA * critic(S2).squeeze(1) * (1 - T)   # TD 타깃 (끝난 상태면 미래 0)
    critic_loss = F.mse_loss(v, target)                        # Critic: V(s)를 타깃에 맞춤
    advantage = (target - v).detach()                          # δ = 이점 추정 (Actor 쪽엔 상수로)
    log_prob = Categorical(logits=actor(S)).log_prob(A)
    actor_loss = -(log_prob * advantage).mean()                # 이점이 +면 그 행동 확률↑

    opt_c.zero_grad(); critic_loss.backward(); opt_c.step()
    opt_a.zero_grad(); actor_loss.backward(); opt_a.step()

    returns_log.append(R.sum().item())
    if (ep + 1) % 100 == 0:
        print(f"episode {ep + 1:4d} | 최근 50판 평균 {np.mean(returns_log[-50:]):6.1f}")

np.savetxt("results/actor_critic.txt", np.array(returns_log))
```

```python
"""10. 가치 반복 · 정책 반복 (전이 확률을 아는 MDP를 동적 계획법으로 풀기) — FrozenLake-v1 (4×4, 미끄러움)
환경의 전이 확률을 그대로 쓴다:  env.unwrapped.P[s][a] = [(확률, 다음 상태, 보상, 끝났나), ...]
실행: python 10_value_iteration.py
"""
import numpy as np
import gymnasium as gym

GAMMA, THETA = 0.99, 1e-8                         # 할인율, 수렴 판정 기준

env = gym.make("FrozenLake-v1", map_name="4x4", is_slippery=True)
P = env.unwrapped.P
nS, nA = env.observation_space.n, env.action_space.n   # 16, 4 (0=왼 1=아래 2=오른 3=위)


def q_from_v(V, s):
    """상태 s에서 행동별 Q(s, a) = Σ p · [r + γ V(s′)]  (끝나는 전이면 미래 가치 0)"""
    q = np.zeros(nA)
    for a in range(nA):
        for prob, s2, r, done in P[s][a]:
            q[a] += prob * (r + GAMMA * V[s2] * (not done))
    return q


def value_iteration():
    V = np.zeros(nS)
    it = 0
    while True:
        it += 1
        delta = 0.0
        for s in range(nS):
            v_new = q_from_v(V, s).max()                  # 벨만 최적 방정식: 행동 중 최댓값
            delta = max(delta, abs(v_new - V[s]))
            V[s] = v_new                                  # 제자리 갱신 (바로 다음 상태 계산에 씀)
        if delta < THETA:                                 # 가장 크게 바뀐 값도 아주 작으면 수렴
            break
    pi = np.array([q_from_v(V, s).argmax() for s in range(nS)])   # V*에서 greedy 정책 뽑기
    return V, pi, it


def policy_evaluation(pi, V):
    """정책 π를 고정하고 V^π 계산: 벨만 기대 방정식을 수렴할 때까지 반복"""
    while True:
        delta = 0.0
        for s in range(nS):
            v_new = q_from_v(V, s)[pi[s]]                 # 결정적 정책: Σ_a π(a|s) Q(s,a) = Q(s, π(s))
            delta = max(delta, abs(v_new - V[s]))
            V[s] = v_new
        if delta < THETA:
            return V


def policy_iteration():
    pi = np.zeros(nS, dtype=int)                          # 아무 정책(전부 '왼쪽')에서 시작
    V = np.zeros(nS)
    rounds = 0
    while True:
        rounds += 1
        V = policy_evaluation(pi, V)                      # ① 평가
        new_pi = np.array([q_from_v(V, s).argmax() for s in range(nS)])   # ② 개선 (greedy)
        if np.array_equal(new_pi, pi):                    # 정책이 더 안 바뀌면 최적
            return V, pi, rounds
        pi = new_pi


def success_rate(pi, episodes=2000, seed=0):
    """정책대로 실제로 해 보고 목표 도착 비율 재기"""
    e = gym.make("FrozenLake-v1", map_name="4x4", is_slippery=True)
    wins = 0.0
    for i in range(episodes):
        s, _ = e.reset(seed=seed + i)
        done = False
        while not done:
            s, r, term, trunc, _ = e.step(int(pi[s]))
            done = term or trunc
        wins += r                                         # 목표 도착 때만 r = 1
    return wins / episodes


if __name__ == "__main__":
    V1, pi1, it = value_iteration()
    V2, pi2, rounds = policy_iteration()
    arrows = np.array(["←", "↓", "→", "↑"])
    desc = ["".join(c.decode() for c in row) for row in env.unwrapped.desc]
    print("지도 (S 출발, F 얼음, H 구멍, G 목표):", " / ".join(desc))
    print(f"가치 반복: {it}번 반복 / 정책 반복: {rounds}번 (평가 → 개선)")
    print(f"두 방법 V 최대 차이 {np.abs(V1 - V2).max():.1e}, 정책 같음: {np.array_equal(pi1, pi2)}")
    print("V*:\n", V1.reshape(4, 4).round(3))
    print("π*:\n", arrows[pi1].reshape(4, 4))
    print(f"최적 정책 성공률(2000판): {success_rate(pi1):.3f}  /  무작위가 아닌 '전부 오른쪽' 정책: {success_rate(np.full(nS, 2)):.3f}")
```

```python
"""11. 탐험-활용: 10개 팔 밴딧에서 greedy · ε-greedy · UCB 비교 (Sutton & Barto 2장 실험과 같은 설정)
실행: python 11_bandit.py
"""
import numpy as np

K, STEPS, RUNS = 10, 1000, 300                    # 팔 10개, 1000번 당기기, 300번 반복 평균


def argmax_random(x, rng):
    """최댓값이 여러 개면 그중 무작위 (항상 첫 번째를 고르면 편향)"""
    return int(rng.choice(np.flatnonzero(x == x.max())))


def run(method, eps=0.0, c=2.0, seed=0):
    rng = np.random.default_rng(seed)
    avg_reward, best_ratio = np.zeros(STEPS), np.zeros(STEPS)
    for _ in range(RUNS):
        q_true = rng.normal(0, 1, K)              # 팔마다 진짜 평균 보상 (에이전트는 모름)
        best = int(np.argmax(q_true))
        Q, N = np.zeros(K), np.zeros(K)           # 추정 가치, 당긴 횟수
        for t in range(1, STEPS + 1):
            if method == "ucb":
                if (N == 0).any():
                    a = int(np.flatnonzero(N == 0)[0])                # 안 당겨 본 팔부터 한 번씩
                else:
                    a = argmax_random(Q + c * np.sqrt(np.log(t) / N), rng)   # 덜 당긴 팔에 가산점
            else:
                a = int(rng.integers(K)) if rng.random() < eps else argmax_random(Q, rng)
            r = rng.normal(q_true[a], 1)          # 보상 = 진짜 평균 + 잡음
            N[a] += 1
            Q[a] += (r - Q[a]) / N[a]             # 증분 평균: 새 평균 = 옛 평균 + (새 값 − 옛 평균)/n
            avg_reward[t - 1] += r
            best_ratio[t - 1] += (a == best)
    return avg_reward / RUNS, best_ratio / RUNS


if __name__ == "__main__":
    results = {}
    for name, kw in [("greedy (ε=0)", dict(method="eps", eps=0.0)), ("ε=0.01", dict(method="eps", eps=0.01)),
                     ("ε=0.1", dict(method="eps", eps=0.1)), ("UCB c=2", dict(method="ucb", c=2.0))]:
        r, b = run(**kw)
        results[name] = (r, b)
        print(f"{name:13s} | 마지막 100번 평균 보상 {r[-100:].mean():.3f} | 최적 팔 고른 비율 {b[-100:].mean() * 100:5.1f}%")
    np.savez("results/bandit.npz", **{k.replace(" ", "_"): np.stack(v) for k, v in results.items()})
```

```python
"""12. A* 격자 경로 탐색 (+ 휴리스틱 0 = Dijkstra와 비교) — 표준 라이브러리 heapq만 사용
실행: python 12_astar.py
"""
import heapq

GRID = [
    "S....#....",
    ".##..#.##.",
    "..#....#..",
    "..#.####..",
    "..#....#..",
    "..####.#..",
    ".......#.G",
]
ROWS, COLS = len(GRID), len(GRID[0])


def find(ch):
    for r, row in enumerate(GRID):
        if ch in row:
            return (r, row.index(ch))


def neighbors(node):
    r, c = node
    for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:                  # 상하좌우 4방향
        nr, nc = r + dr, c + dc
        if 0 <= nr < ROWS and 0 <= nc < COLS and GRID[nr][nc] != "#":
            yield (nr, nc)


def manhattan(a, b):
    return abs(a[0] - b[0]) + abs(a[1] - b[1])        # 4방향 이동에서 실제 비용보다 크지 않음(admissible)


def astar(start, goal, h):
    open_heap = [(h(start, goal), 0, start)]           # (f = g + h, g, 노드) — f가 작은 것부터 꺼냄
    g = {start: 0}                                      # 출발점에서 각 노드까지 지금까지 찾은 최소 비용
    parent = {start: None}
    closed = set()                                      # 확정된(다시 볼 필요 없는) 노드
    expanded = 0
    while open_heap:
        f, g_cur, node = heapq.heappop(open_heap)
        if node in closed:                              # 더 비싼 옛 기록이면 건너뜀
            continue
        closed.add(node)
        expanded += 1
        if node == goal:
            break
        for nb in neighbors(node):
            ng = g_cur + 1                              # 한 칸 이동 비용 1
            if ng < g.get(nb, float("inf")):            # 더 싼 길을 찾았으면 갱신
                g[nb] = ng
                parent[nb] = node
                heapq.heappush(open_heap, (ng + h(nb, goal), ng, nb))
    if goal not in parent:
        return None, expanded
    path, node = [], goal
    while node is not None:                             # 부모를 따라 거꾸로 올라감
        path.append(node)
        node = parent[node]
    return path[::-1], expanded


if __name__ == "__main__":
    S, G = find("S"), find("G")
    path_a, exp_a = astar(S, G, manhattan)
    path_d, exp_d = astar(S, G, lambda a, b: 0)        # h = 0 → Dijkstra
    print(f"A*      : 경로 {len(path_a) - 1}칸, 펼친 노드 {exp_a}개")
    print(f"Dijkstra: 경로 {len(path_d) - 1}칸, 펼친 노드 {exp_d}개")
    grid = [list(row) for row in GRID]
    for (r, c) in path_a[1:-1]:
        grid[r][c] = "*"
    print("\n".join("".join(row) for row in grid))
```

```python
"""00. gymnasium 기본 API + 내 환경 만들기 + 보상 Wrapper
실행: python 00_gym_basics.py
"""
import numpy as np
import gymnasium as gym
from gymnasium import spaces
from gymnasium.utils.env_checker import check_env

# ── 1) 이미 있는 환경을 무작위 행동으로 한 판 돌리기 ──────────────────
env = gym.make("CartPole-v1")
print("관측 공간:", env.observation_space)       # Box(4,) : 카트 위치·속도, 막대 각도·각속도
print("행동 공간:", env.action_space)            # Discrete(2) : 0=왼쪽, 1=오른쪽
obs, info = env.reset(seed=0)                    # reset은 (관측, 정보) 두 개를 돌려줌
total, steps = 0.0, 0
while True:
    action = env.action_space.sample()           # 무작위 행동
    obs, reward, terminated, truncated, info = env.step(action)   # step은 5개를 돌려줌
    total += reward
    steps += 1
    if terminated or truncated:                  # terminated=실패/성공으로 끝, truncated=시간 제한으로 잘림
        break
print(f"무작위 정책: {steps}스텝, 보상 합 {total}")
env.close()


# ── 2) 내 환경을 gym.Env 규칙대로 만들기 ─────────────────────────────
class GridEnv(gym.Env):
    """n×n 격자. (0,0)에서 출발해 (n-1,n-1)에 도착하면 성공."""

    def __init__(self, n=5, max_steps=50):
        super().__init__()
        self.n, self.max_steps = n, max_steps
        self.observation_space = spaces.Box(0, n - 1, shape=(2,), dtype=np.float32)
        self.action_space = spaces.Discrete(4)
        self.moves = np.array([[-1, 0], [1, 0], [0, -1], [0, 1]])   # 위·아래·왼·오
        self.goal = np.array([n - 1, n - 1])

    def reset(self, seed=None, options=None):
        super().reset(seed=seed)                  # self.np_random 시드 설정
        self.pos = np.array([0, 0])
        self.t = 0
        return self.pos.astype(np.float32), {}

    def step(self, action):
        self.pos = np.clip(self.pos + self.moves[action], 0, self.n - 1)   # 벽 밖으로 못 나감
        self.t += 1
        terminated = bool((self.pos == self.goal).all())                  # 목표 도착 = 진짜 끝
        truncated = self.t >= self.max_steps                               # 시간 초과 = 잘림
        reward = 1.0 if terminated else -0.01                              # 도착 +1, 매 스텝 −0.01
        return self.pos.astype(np.float32), reward, terminated, truncated, {}


check_env(GridEnv())                              # 규칙(반환 형식·공간)을 지켰는지 자동 검사
print("GridEnv check_env 통과")


# ── 3) Wrapper로 보상만 바꾸기 (환경 코드는 그대로) ──────────────────
class DistanceShaping(gym.Wrapper):
    """잠재 기반 보상 설계: r' = r + γΦ(s') − Φ(s),  Φ(s) = −(목표까지 맨해튼 거리)"""

    def __init__(self, env, gamma=0.99, scale=0.1):
        super().__init__(env)
        self.gamma, self.scale = gamma, scale

    def phi(self, obs):
        return -self.scale * np.abs(obs - self.env.unwrapped.goal).sum()

    def reset(self, **kwargs):
        obs, info = self.env.reset(**kwargs)
        self.last_phi = self.phi(obs)
        return obs, info

    def step(self, action):
        obs, r, terminated, truncated, info = self.env.step(action)
        phi_next = 0.0 if terminated else self.phi(obs)      # 끝난 상태의 Φ는 0
        shaped = r + self.gamma * phi_next - self.last_phi
        self.last_phi = phi_next
        info["original_reward"] = r                           # 원래 보상은 따로 기록 (평가용)
        return obs, shaped, terminated, truncated, info


env = DistanceShaping(GridEnv())
obs, _ = env.reset(seed=0)
obs, r, term, trunc, info = env.step(3)                       # 오른쪽 = 목표 쪽으로 한 칸
print(f"목표 쪽으로 한 칸: 설계 보상 {r:+.3f} (원래 {info['original_reward']:+.2f})")
obs, r, term, trunc, info = env.step(2)                       # 왼쪽 = 목표 반대
print(f"반대로 한 칸:     설계 보상 {r:+.3f} (원래 {info['original_reward']:+.2f})")

# 상태가 필요 없는 보상 변환은 RewardWrapper가 더 간단
class ScaleReward(gym.RewardWrapper):
    def reward(self, reward):
        return 0.1 * reward
```

```python
# ── R1 ε-greedy · Q-learning · SARSA ──
def eps_greedy(Q, s, eps, rng):
    return int(rng.integers(Q.shape[1])) if rng.random() < eps else int(np.argmax(Q[s]))


def q_learning_update(Q, s, a, r, s2, done, alpha=0.1, gamma=0.99):
    target = r + gamma * np.max(Q[s2]) * (not done)
    Q[s, a] += alpha * (target - Q[s, a])


def sarsa_update(Q, s, a, r, s2, a2, done, alpha=0.1, gamma=0.99):
    target = r + gamma * Q[s2, a2] * (not done)
    Q[s, a] += alpha * (target - Q[s, a])


# ── R2 할인 리턴 (REINFORCE) ──
def discounted_returns(rewards, gamma=0.99):
    G, out = 0.0, []
    for r in reversed(rewards):
        G = r + gamma * G
        out.insert(0, G)
    return out


# ── R3 DQN 손실 ──
def dqn_loss(q, q_target, s, a, r, s2, done, gamma=0.99, double=True):
    q_sa = q(s).gather(1, a.unsqueeze(1)).squeeze(1)
    with torch.no_grad():
        if double:
            a2 = q(s2).argmax(dim=1, keepdim=True)
            next_q = q_target(s2).gather(1, a2).squeeze(1)
        else:
            next_q = q_target(s2).max(dim=1).values
        y = r + gamma * (1 - done) * next_q
    return F.smooth_l1_loss(q_sa, y)


# ── R4 Actor-Critic 손실 ──
def actor_critic_losses(actor, critic, s, a, r, s2, done, gamma=0.99):
    v = critic(s).squeeze(-1)
    with torch.no_grad():
        target = r + gamma * critic(s2).squeeze(-1) * (1 - done)
    delta = (target - v).detach()
    logp = Categorical(logits=actor(s)).log_prob(a)
    return -(logp * delta).mean(), F.mse_loss(v, target)


# ── R5 GAE ──
def gae(rewards, values, dones, last_value, gamma=0.99, lam=0.95):
    T = len(rewards)
    adv, g = [0.0] * T, 0.0
    for t in reversed(range(T)):
        next_v = last_value if t == T - 1 else values[t + 1]
        delta = rewards[t] + gamma * next_v * (1 - dones[t]) - values[t]
        g = delta + gamma * lam * (1 - dones[t]) * g
        adv[t] = g
    returns = [adv[t] + values[t] for t in range(T)]
    return adv, returns


# ── R6 PPO 손실 ──
def ppo_loss(new_logp, old_logp, adv, values, returns, entropy, clip=0.2, c1=0.5, c2=0.01):
    ratio = torch.exp(new_logp - old_logp)
    adv = (adv - adv.mean()) / (adv.std() + 1e-8)
    policy_loss = -torch.min(ratio * adv, torch.clamp(ratio, 1 - clip, 1 + clip) * adv).mean()
    value_loss = ((values - returns) ** 2).mean()
    return policy_loss + c1 * value_loss - c2 * entropy.mean()


# ── R7 SAC: tanh 가우시안 정책 · 손실 · Polyak ──
def tanh_gaussian(mu, log_std, scale=1.0):
    dist = Normal(mu, log_std.clamp(-20, 2).exp())
    u = dist.rsample()
    y = torch.tanh(u)
    logp = dist.log_prob(u).sum(-1) - torch.log(scale * (1 - y.pow(2)) + 1e-6).sum(-1)
    return y * scale, logp


def sac_losses(pi, q1, q2, q1_t, q2_t, s, a, r, s2, done, alpha, gamma=0.99):
    with torch.no_grad():
        a2, logp2 = pi(s2)
        y = r + gamma * (1 - done) * (torch.min(q1_t(s2, a2), q2_t(s2, a2)) - alpha * logp2)
    q_loss = F.mse_loss(q1(s, a), y) + F.mse_loss(q2(s, a), y)
    a_new, logp = pi(s)
    pi_loss = (alpha * logp - torch.min(q1(s, a_new), q2(s, a_new))).mean()
    return q_loss, pi_loss, logp


def alpha_loss(log_alpha, logp, target_entropy):
    return -(log_alpha.exp() * (logp.detach() + target_entropy)).mean()


def soft_update(net, target, tau=0.005):
    with torch.no_grad():
        for p, pt in zip(net.parameters(), target.parameters()):
            pt.mul_(1 - tau).add_(tau * p)


# ── R8 가치 반복 ──
def value_iteration(P, nS, nA, gamma=0.99, theta=1e-8):
    V = np.zeros(nS)
    q = lambda s, a: sum(p * (r + gamma * V[s2] * (not d)) for p, s2, r, d in P[s][a])
    while True:
        delta = 0.0
        for s in range(nS):
            best = max(q(s, a) for a in range(nA))
            delta = max(delta, abs(best - V[s]))
            V[s] = best
        if delta < theta:
            break
    pi = [max(range(nA), key=lambda a: q(s, a)) for s in range(nS)]
    return V, pi

def ucb_action(Q, N, t, c=2.0):
    if (N == 0).any():
        return int(np.argmin(N))
    return int(np.argmax(Q + c * np.sqrt(np.log(t) / N)))

    Q[a] += (r - Q[a]) / N[a]

# ── R11 RRT ──
def rrt(start, goal, is_free, lo, hi, step=0.5, goal_bias=0.1, tol=0.5, iters=5000, seed=0):
    rng = np.random.default_rng(seed)
    nodes, parent = [np.array(start, float)], [-1]
    goal = np.array(goal, float)
    for _ in range(iters):
        x_rand = goal if rng.random() < goal_bias else rng.uniform(lo, hi)
        i = int(np.argmin([np.linalg.norm(n - x_rand) for n in nodes]))
        d = x_rand - nodes[i]
        dist = np.linalg.norm(d)
        if dist < 1e-9:
            continue
        x_new = nodes[i] + d / dist * min(step, dist)
        if not is_free(nodes[i], x_new):
            continue
        nodes.append(x_new)
        parent.append(i)
        if np.linalg.norm(x_new - goal) <= tol and is_free(x_new, goal):
            path, k = [goal], len(nodes) - 1
            while k != -1:
                path.append(nodes[k])
                k = parent[k]
            return path[::-1]
    return None


# ── R12 A* ──
def astar(start, goal, neighbors, h):
    heap, g, parent, closed = [(h(start), 0, start)], {start: 0}, {start: None}, set()
    while heap:
        f, gc, u = heapq.heappop(heap)
        if u in closed:
            continue
        closed.add(u)
        if u == goal:
            break
        for v in neighbors(u):
            if gc + 1 < g.get(v, float("inf")):
                g[v], parent[v] = gc + 1, u
                heapq.heappush(heap, (gc + 1 + h(v), gc + 1, v))
    if goal not in parent:
        return None
    path, u = [], goal
    while u is not None:
        path.append(u)
        u = parent[u]
    return path[::-1]
```

```python
"""3교시 예상 문제 전수 목록(06)에 실리는 추가 코드. 검증: python extra_snippets_check.py"""
import numpy as np
import torch
import torch.nn as nn


# ── X1 Dueling DQN 머리: Q = V + (A − A의 평균) ──
class DuelingQ(nn.Module):
    def __init__(self, obs_dim, n_act, h=128):
        super().__init__()
        self.body = nn.Sequential(nn.Linear(obs_dim, h), nn.ReLU())
        self.v = nn.Linear(h, 1)           # 상태 가치 V(s)
        self.a = nn.Linear(h, n_act)       # 행동별 이점 A(s, a)

    def forward(self, x):
        z = self.body(x)
        a = self.a(z)
        return self.v(z) + a - a.mean(dim=1, keepdim=True)


# ── X2 n-step 목표값 ──
def n_step_target(rewards, gamma, bootstrap_value, done):
    """rewards = [r_t, …, r_(t+n−1)], bootstrap_value = n스텝 뒤 상태의 가치 추정(max Q 또는 V)"""
    G = sum(gamma ** k * r for k, r in enumerate(rewards))
    if not done:
        G += gamma ** len(rewards) * bootstrap_value
    return G


# ── X3 행동 복제(behavior cloning): 시연 (상태, 행동)으로 지도학습 ──
def behavior_cloning(states, actions, n_act, epochs=200, lr=1e-2):
    policy = nn.Sequential(nn.Linear(states.shape[1], 64), nn.ReLU(), nn.Linear(64, n_act))
    opt = torch.optim.Adam(policy.parameters(), lr=lr)
    loss_fn = nn.CrossEntropyLoss()          # 연속 행동이면 nn.MSELoss()
    for _ in range(epochs):
        opt.zero_grad()
        loss = loss_fn(policy(states), actions)
        loss.backward()
        opt.step()
    return policy


# ── X4 GRPO 그룹 상대 이점: 같은 과제를 여러 번 시도한 보상들을 그룹 안에서 정규화 ──
def grpo_advantage(group_rewards):
    r = np.asarray(group_rewards, dtype=float)
    return (r - r.mean()) / (r.std() + 1e-8)


# ── X5 하위 목표 숙련도·보상 가중치 (Agentic-VLA 보상 합성 식) ──
def update_subgoal_weights(c, success, alpha=0.9):
    """c_k ← α·c_k + (1 − α)·성공(0/1),  가중치 w_k = 1 − c_k"""
    c = alpha * c + (1 - alpha) * success
    return c, 1 - c


# ── X6 경로 다듬기(shortcut): 두 점을 직선으로 이어도 충돌이 없으면 중간 점을 지움 ──
def shortcut(path, edge_free, iters=300, seed=0):
    rng = np.random.default_rng(seed)
    path = [np.asarray(p, dtype=float) for p in path]
    for _ in range(iters):
        if len(path) < 3:
            break
        i, j = sorted(rng.choice(len(path), 2, replace=False))
        if j - i >= 2 and edge_free(path[i], path[j]):
            path = path[:i + 1] + path[j:]
    return path


# ── X7 학습 곡선 이동평균 (출렁이는 리턴을 부드럽게 보기) ──
def moving_average(x, w=20):
    x = np.asarray(x, dtype=float)
    return np.convolve(x, np.ones(w) / w, mode="valid")
```
